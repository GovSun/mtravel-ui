/** Reads --sheet-motion-* from CSS on the sheet itself, so Storybook's slow-motion / reduced-motion reach the JS. */
function readConfig(el: HTMLElement): MotionConfig {
  const m = readMotion(el);
  const num = (token: string, fallback: number) => {
    const n = parseFloat(m.value(token, ''));
    return Number.isFinite(n) ? n : fallback;
  };
  return {
    openDuration: m.duration('--sheet-motion-open-duration', DEFAULT_CONFIG.openDuration),
    closeDuration: m.duration('--sheet-motion-close-duration', DEFAULT_CONFIG.closeDuration),
    easing: parseCubicBezier(m.value('--sheet-motion-easing', '')) ?? DEFAULT_CONFIG.easing,
    springResponse: m.duration('--sheet-motion-spring-response', DEFAULT_CONFIG.springResponse),
    springDamping: num('--sheet-motion-spring-damping', DEFAULT_CONFIG.springDamping),
    flickDamping: num('--sheet-motion-flick-damping', DEFAULT_CONFIG.flickDamping),
    exitAccel: num('--sheet-motion-exit-accel', DEFAULT_CONFIG.exitAccel),
    exitSpeed: num('--sheet-motion-exit-speed', DEFAULT_CONFIG.exitSpeed),
    scale: num('--motion-scale', 1),
    fadeDuration: m.duration('--sheet-motion-fade-duration', DEFAULT_CONFIG.fadeDuration),
    reduce: m.reduce,
  };
}

 * BottomSheet (Figma 162:687): a modal sheet from the bottom edge that moves like a native iOS sheet.
 *
 * Motion lives in `engine.ts` (tween on the drawer curve from rest, a velocity-carrying spring for anything
 * the user touched) and `gesture.ts` (1:1 drag, scroll hand-off, rubber band, flick). React only decides
 * when the sheet is mounted: it stays in the tree while it slides out and unmounts when it has left.
 */
export function BottomSheet({
  open,
  defaultOpen = false,
  onOpenChange,
  header,
  title,
  description,
  'aria-label': ariaLabel,
  showHandle = true,
  showClose = true,
  closeLabel = 'Закрыть',
  dismissible = true,
  primaryAction,
  secondaryAction,
  footer,
  initialFocus,
  children,
  className,
  onOpened,
  onClosed,
}: BottomSheetProps) {
  const [isOpen, setOpen] = useControllableState(open, defaultOpen, onOpenChange);
  const [present, setPresent] = useState(isOpen);
  if (isOpen && !present) setPresent(true);

  const titleId = useId();
  const sheetRef = useRef<HTMLDivElement>(null);
  const backdropRef = useRef<HTMLDivElement>(null);
  const closeRef = useRef<HTMLButtonElement>(null);
  const contentRef = useRef<HTMLDivElement>(null);
  const motionRef = useRef<SheetMotion | null>(null);
  const [overflowing, setOverflowing] = useState(false);

  // always-current values for the long-lived engine and gesture callbacks
  const live = useRef({ isOpen, dismissible, onOpened, onClosed, setOpen });
  live.current = { isOpen, dismissible, onOpened, onClosed, setOpen };

  const resolvedHeader: BottomSheetHeader = header ?? (title ? 'left' : 'none');
  const hasFooter = footer != null || primaryAction != null || secondaryAction != null;

  // Closing starts inside the same event (no wait for React): the sheet moves on the very next frame.
  // A controlled parent may refuse; two frames later a sheet that is still "open" reverses from where it is.
  const requestClose = useCallback(() => {
    if (!live.current.isOpen) return;
    const m = motionRef.current;
    const sheet = sheetRef.current;
    if (m && sheet && m.goal === 'open') {
      m.cfg = readConfig(sheet);
      m.go('closed');
    }
    live.current.setOpen(false);
    requestAnimationFrame(() =>
      requestAnimationFrame(() => {
        const cur = motionRef.current;
        if (cur && sheet && live.current.isOpen && cur.goal === 'closed') {
          cur.cfg = readConfig(sheet);
          cur.go('open');
        }
      }),
    );
  }, []);

  /* ---------- motion: engine + gesture, created when the sheet is mounted ---------- */
  useLayoutEffect(() => {
    if (!present) return;
    const sheet = sheetRef.current;
    const backdrop = backdropRef.current;
    if (!sheet || !backdrop) return;

    const m = new SheetMotion(sheet, backdrop, readConfig(sheet));
    motionRef.current = m;
    m.H = sheet.offsetHeight;
    m.reset(true);
    m.onSettle = (goal) => {
      if (goal === 'open') {
        live.current.onOpened?.();
      } else if (live.current.isOpen) {
        // closed by a gesture, but the parent kept it open: come back
        m.cfg = readConfig(sheet);
        m.go('open');
      } else {
        motionRef.current = null;
        setPresent(false);
        live.current.onClosed?.();
      }
    };

    const gesture = new SheetGesture(sheet, {
      motion: () => motionRef.current,
      dismissible: () => live.current.dismissible,
      isOpen: () => live.current.isOpen,
      requestClose,
      requestOpen: () => live.current.setOpen(true),
      refresh: () => {
        if (motionRef.current) motionRef.current.cfg = readConfig(sheet);
      },
    });

    // content height changes, rotation, keyboard: the cached height is the only thing the frame loop reads
    const ro = new ResizeObserver(() => m.setHeight(sheet.offsetHeight));
    ro.observe(sheet);

    // a scrollable slot must be reachable from the keyboard: it becomes a focusable region only while it overflows
    const content = contentRef.current;
    const measure = () => content && setOverflowing(content.scrollHeight > content.clientHeight + 1);
    const ro2 = new ResizeObserver(measure);
    const mo = new MutationObserver(() => {
      measure();
      if (content) for (const c of Array.from(content.children)) ro2.observe(c);
    });
    if (content) {
      ro2.observe(content);
      for (const c of Array.from(content.children)) ro2.observe(c);
      mo.observe(content, { childList: true });
    }
    measure();

    m.go(live.current.isOpen ? 'open' : 'closed');

    return () => {
      ro2.disconnect();
      mo.disconnect();
      ro.disconnect();
      gesture.destroy();
      m.destroy();
      if (motionRef.current === m) motionRef.current = null;
    };
  }, [present, requestClose]);

  // open <-> closed from the outside (props, ✕, Escape, backdrop): reverses from wherever the sheet is
  useLayoutEffect(() => {
    const m = motionRef.current;
    const sheet = sheetRef.current;
    if (!m || !sheet) return;
    if ((m.goal === 'open') === isOpen) return;
    m.cfg = readConfig(sheet);
    m.go(isOpen ? 'open' : 'closed');
  }, [isOpen]);

  /* ---------- page lock: scroll lock + inert background live as long as the sheet is on screen, released when it has left ---------- */
  const inertRef = useRef<(() => void) | null>(null);
  useEffect(() => {
    if (!present) return;
    const root = sheetRef.current?.parentElement;
    if (!root) return;
    const unlock = lockScroll();
    const uninert = inertOthers(root);
    inertRef.current = uninert;
    return () => {
      inertRef.current = null;
      uninert();
      unlock();
    };
  }, [present]);

  /* ---------- modal behaviour: focus, Escape, focus return (as soon as closing starts) ---------- */
  useEffect(() => {
    if (!isOpen || !present) return;
    const sheet = sheetRef.current;
    const root = sheet?.parentElement;
    if (!sheet || !root) return;

    const trigger = document.activeElement instanceof HTMLElement ? document.activeElement : null;
    const id = Symbol('sheet');
    sheetStack.push(id);

    const target =
      typeof initialFocus === 'string'
        ? initialFocus === 'close'
          ? closeRef.current
          : sheet.querySelector<HTMLElement>(initialFocus)
        : initialFocus?.current;
    (target ?? sheet).focus({ preventScroll: true });

    const onKey = (e: globalThis.KeyboardEvent) => {
      if (e.key === 'Escape' && sheetStack.isTop(id) && live.current.dismissible && !e.defaultPrevented) {
        e.preventDefault();
        requestClose();
      }
    };
    const onFocusIn = (e: FocusEvent) => {
      if (sheetStack.isTop(id) && e.target instanceof Node && !root.contains(e.target)) (sheet.querySelector<HTMLElement>(FOCUSABLE) ?? sheet).focus({ preventScroll: true });
    };
    document.addEventListener('keydown', onKey);
    document.addEventListener('focusin', onFocusIn);

    return () => {
      document.removeEventListener('keydown', onKey);
      document.removeEventListener('focusin', onFocusIn);
      sheetStack.remove(id);
      if (trigger && trigger.isConnected && trigger !== document.body) {
        // the page stays inert until the sheet has left; the one thing that must be focusable now is the trigger
        const top = trigger.closest('body > *') as (HTMLElement & { inert?: boolean }) | null;
        if (top && top !== root) {
          top.inert = false;
          top.removeAttribute('aria-hidden');
        }
        trigger.focus({ preventScroll: true });
      }
    };
  }, [isOpen, present, initialFocus, requestClose]);

  const onKeyDown = (e: KeyboardEvent<HTMLDivElement>) => {
    if (e.key !== 'Tab') return;
    const sheet = sheetRef.current;
    if (!sheet) return;
    const items = Array.from(sheet.querySelectorAll<HTMLElement>(FOCUSABLE)).filter((el) => el.tabIndex >= 0 && !el.hasAttribute('inert'));
    const first = items[0];
    const last = items[items.length - 1];
    if (!first || !last) {
      e.preventDefault();
      return;
    }
    const active = document.activeElement;
    if (e.shiftKey && (active === first || active === sheet)) {
      e.preventDefault();
      last.focus();
    } else if (!e.shiftKey && active === last) {
      e.preventDefault();
      first.focus();
    }
  };

  if (!present || typeof document === 'undefined') return null;

  const closeSize = resolvedHeader === 'none' ? 40 : 36;
  const named = resolvedHeader !== 'none' && title ? { 'aria-labelledby': titleId } : { 'aria-label': ariaLabel ?? (typeof title === 'string' ? title : undefined) };

  return createPortal(
    <div className={styles.root} data-state={isOpen ? 'open' : 'closed'}>
      <div
        ref={backdropRef}
        className={styles.backdrop}
        style={{ opacity: 0 }}
        onClick={() => live.current.dismissible && requestClose()}
        aria-hidden="true"
        data-sheet-backdrop=""
      />
      <div
        ref={sheetRef}
        className={[styles.sheet, className].filter(Boolean).join(' ')}
        style={{ transform: 'translate3d(0, 100%, 0)' }}
        role="dialog"
        aria-modal="true"
        tabIndex={-1}
        onKeyDown={onKeyDown}
        data-header={resolvedHeader}
        data-handle={showHandle || undefined}
        data-close={showClose || undefined}
        data-footer={hasFooter || undefined}
        {...named}
      >
        <div className={styles.panel}>
          {showHandle && (
            <div className={styles.handle} aria-hidden="true">
              <span className={styles.bar} />
            </div>
          )}
          {showClose && (
            <IconButton
              ref={closeRef}
              className={styles.close}
              icon={<CloseIcon />}
              aria-label={closeLabel}
              variant="ghost"
              radius={16}
              size={closeSize}
              onClick={requestClose}
            />
          )}
          {resolvedHeader === 'none' ? (
            showClose && <div className={styles.topbar} aria-hidden="true" />
          ) : (
            <header className={styles.header}>
              <div className={styles.headerRow}>
                {title && (
                  <h2 id={titleId} className={`im-24-b ${styles.title}`}>
                    {title}
                  </h2>
                )}
                {description && <p className={`im-16-r ${styles.description}`}>{description}</p>}
              </div>
            </header>
          )}
          <div
            ref={contentRef}
            className={styles.content}
            data-sheet-content=""
            {...(overflowing
              ? { tabIndex: 0, role: 'region', 'aria-label': typeof title === 'string' ? title : (ariaLabel ?? 'Содержимое') }
              : {})}
          >
            {children}
          </div>
          {hasFooter && (
            <footer className={styles.footer}>
              {footer ?? (
                <>
                  {primaryAction && (
                    <Button fullWidth size={52} variant="primary" loading={primaryAction.loading} disabled={primaryAction.disabled} onClick={primaryAction.onClick}>
                      {primaryAction.label}
                    </Button>
                  )}
                  {secondaryAction && (
                    <Button fullWidth size={52} variant="secondary" loading={secondaryAction.loading} disabled={secondaryAction.disabled} onClick={secondaryAction.onClick}>
                      {secondaryAction.label}
                    </Button>
                  )}
                </>
              )}
            </footer>
          )}
        </div>
      </div>
    </div>,
    document.body,
  );
}

# BottomSheet

Шторка снизу: выбор параметров, подтверждения, короткие формы (пассажиры, фильтры, условия). Движется как нативная шторка iOS: палец и шторка
двигаются 1:1, бросок продолжается с той же скоростью, шторку можно схватить в любой момент анимации. Работает в React-webview на iPhone и Android.

Figma: [Bottom sheet 162:687](https://www.figma.com/design/Enj0CUGKB1uJvXiWt7Vu8P/Design-System---Mtravel-2.0?node-id=162-687) (Header: None / Top left / Top center).

Шторка рендерится в `document.body` и остаётся в дереве, пока уезжает вниз. `onOpenChange(false)` приходит от ✕, тапа по фону, Escape и свайпа вниз;
`onOpenChange(true)` — если пользователь поймал уезжающую шторку и отпустил её обратно.

## Движение

Всё движение шторки считает собственный движок (`engine.ts`, `gesture.ts`), без зависимостей. CSS-анимации для жестов не годятся: их нельзя схватить и развернуть
на лету. Шторка двигается только через `transform` и `opacity`, `transform` пишется прямо в элемент (не через CSS-переменную, чтобы не пересчитывать стили всего содержимого),
`will-change` стоит только пока что-то движется.

### Модель

Положение `y` в px от открытого состояния: `0` открыта, `H` целиком ниже экрана (`H` — высота шторки, кэшируется и обновляется через `ResizeObserver`).
Прозрачность фона `= 1 − clamp(y / H, 0, 1)` пересчитывается при каждой записи положения, поэтому фон всегда привязан к шторке.

| Ситуация | Что двигает шторку | Почему |
| --- | --- | --- |
| Открытие из покоя | кривая drawer `cubic-bezier(0.32, 0.72, 0, 1)`, 500 мс | резкий старт и длинная мягкая посадка, как у iOS; вход — ease-out |
| Закрытие (✕, фон, Escape, `open=false`) | та же кривая назад, 400 мс | тот же путь, но быстрее: система отвечает, пользователь не ждёт |
| Уход после жеста, разворот открытия в закрытие | кубическая кривая Эрмита из (y, v₀) в (H, 1500 px/с) с ограниченным стартовым ускорением | шторка продолжает на скорости пальца и набирает скорость плавно (рост не больше ~4 px за кадр на кадр). Пружина, нацеленная на край, давала 100 000 px/с² с первого кадра и бросала шторку быстрее пальца |
| Возврат в открытое положение, захват движущейся шторки | пружина с начальной скоростью пальца | у пружины нет «рестарта», она подхватывает положение и скорость |

Анимация по кривой стартует с экранного положения. Если шторка уже летит быстрее 150 px/с (разворот, захват), берётся Эрмит (на уход) или пружина (на возврат), чтобы скорость не обрывалась. Закрытие начинается синхронно в обработчике ✕, фона и Escape, без ожидания React.

### Уход после жеста

```
y(u) = h00·y₀ + h10·T·v₀ + h01·H + h11·T·v_end,   u = t / T
y''(0) = (6D − T(4v₀ + 2v_end)) / T² = 14 400 px/с²,   D = H − y₀,  v_end = max(v₀, 1500 px/с)
```

`T` решает это уравнение и не превышает 400 мс. Токены `--sheet-motion-exit-accel`, `--sheet-motion-exit-speed`. Если шторка летела вверх (разворот), она сначала гасит эту скорость с тем же ускорением и затем идёт вниз.

### Пружина возврата

Параметры как у Apple: отклик (response) и коэффициент демпфирования (damping ratio), решение в замкнутом виде, поэтому результат не зависит от частоты кадров.

```
ω  = 2π / response           response = 380 мс (токен --sheet-motion-spring-response)
ζ  = 1                       без отскока (по умолчанию)
ζ  = 0.82                    после броска вверх: лёгкий отскок
ζ = 1:  x(t) = (A + (v₀ + ωA)·t) · e^(−ωt)
ζ < 1:  x(t) = e^(−ζωt) · (A·cos ω_d t + (v₀ + ζωA)/ω_d · sin ω_d t),  ω_d = ω·√(1 − ζ²)
```

`A` — расстояние до цели в момент старта, `v₀` — скорость пальца в px/с.

### Что решает при отпускании

- Скорость берётся по последним ~100 мс движения шторки. Если с последнего `touchmove` прошло больше **45 мс** (палец постоял), скорость равна нулю: iOS не шлёт события для неподвижного пальца.
- **Бросок** (скорость вниз ≥ **0.3 px/мс**) закрывает после 8 px пути. **Мягкий бросок** (0.11–0.3 px/мс) закрывает после 48 px. Медленнее — решает только расстояние: дальше **30 %** высоты, либо проекция инерции `y + v/1000 · 0.99/(1 − 0.99)` уходит за 30 %. Движение вверх быстрее 0.11 px/мс всегда возвращает шторку.
- Возврат идёт критически демпфированной пружиной. После броска вверх добавляется отскок `ζ = 0.82`: перелёт есть, потому что у жеста был импульс.

### Резинка

Вверх от открытого положения шторка идёт с растущим сопротивлением: `f(x) = x·L·k / (L + k·x)`, `L = 120 px` (потолок), `k = 0.55` (начальное сопротивление).
Порог начала жеста вычитается, поэтому 100 px пальца дают около трети (33–37 px) движения шторки, жёсткой остановки нет. Щель под приподнятой шторкой закрывает продолжение фона шторки. Если `dismissible=false`, то же сопротивление работает и вниз.

### Reduced motion

При `prefers-reduced-motion` (или «Motion → Reduced motion» в тулбаре) открытие и закрытие становятся короткой сменой прозрачности шторки и фона без
перемещения: длительность **200 мс** по ease-out, заметная часть (от 2 до 98 %) занимает около 170 мс. Перетаскивание остаётся 1:1, потому что его ведёт сам пользователь; короткий жест возвращает шторку пружиной, длинный закрывает её плавным исчезновением.
«Замедление ×4» из тулбара умножает все длительности, в том числе отклик пружины.

## Жесты

| Жест | Поведение |
| --- | --- |
| Тянуть за ручку, заголовок, футер или любую неподвижную область | шторка идёт за пальцем 1:1, с тем смещением, где схватили. Начало жеста — после **4 px** на неподвижных поверхностях и **10 px** внутри прокручиваемого контента, поэтому тапы по кнопкам работают |
| Тянуть по прокручиваемому контенту вниз | пока контент не в самом верху, он прокручивается; дойдя до верха, тот же жест продолжает тянуть шторку. Дальше назад вверх шторка сначала возвращается, потом снова прокручивается контент |
| Тянуть по контенту вверх | прокручивается контент, шторка стоит |
| Горизонтальное движение | игнорируется (карусели внутри контента) |
| Второй палец | игнорируется: ни начать жест, ни перехватить его нельзя |
| Тронуть движущуюся шторку (не кнопку, ссылку или поле: тап по ✕ шторку не останавливает) | она останавливается под пальцем с текущего экранного положения и идёт за ним; если палец поднят без движения, шторка продолжает путь к прежней цели с прежней скоростью |
| Закрывается, а пользователь тянет назад | шторка разворачивается из текущей точки, `onOpenChange(true)` сообщает родителю |
| Мышь и стилус | Pointer Events с `setPointerCapture`; после перетаскивания клик под курсором не срабатывает |

Для пальца используются touch events, а не pointer events: браузер отменяет pointer-события в момент, когда сам начинает прокрутку, и передать жест от прокрутки
шторке внутри одного касания было бы нельзя. Касание неявно захвачено элементом, на котором началось (это эквивалент `setPointerCapture`), а `preventDefault()`
на `touchmove` не даёт странице и WKWebView прокручиваться за шторкой.

## Доступность

- `role="dialog"`, `aria-modal="true"`, имя из заголовка (`aria-labelledby`) или из `aria-label`.
- Слот с прокруткой получает `tabIndex=0`, `role="region"` и `aria-label` только пока контент не помещается, чтобы его можно было прокручивать стрелками; порядок Tab не меняется, фокус при открытии он не забирает.
- Фокус уходит в шторку при открытии (на саму шторку, либо на `initialFocus`) и возвращается на элемент, открывший её, при закрытии.
- Tab и Shift+Tab ходят по кругу внутри шторки; всё остальное на странице получает `inert` и `aria-hidden`. ✕ первый в порядке обхода.
- Escape закрывает (если `dismissible`). Верхняя из нескольких шторок получает Escape первой.
- Блокировка скролла и `inert` снимаются, когда шторка уехала (перед размонтированием, в том числе при размонтировании во время закрытия), а фокус возвращается сразу в начале закрытия.
- Страница под шторкой не прокручивается: блокировка скролла `position: fixed` на `body`, она же держит iOS Safari, позиция прокрутки восстанавливается.
- Кнопка ✕ доступна всегда, даже при `dismissible={false}`.

## Известные ограничения

- Проверено в Chromium с эмуляцией касаний. Ощущение на реальном iPhone зависит от WebKit: проверьте на устройстве, как ведёт себя резинка прокрутки внутри слота.

То же ощущение, что у Counter, токены `--button-motion-press-*`:

- На `pointerdown` фон переключается на токен `…/pressed`, кнопка сжимается до 0.97 (текст и иконки вместе с ней). Обе смены идут за 100 мс по сильному ease-out.

Пять вариантов, проп `loader`. Все линейные варианты двигают только `transform` и `opacity`, так что это дёшево даже в вебвью на слабом Android.

| `loader` | Что видно | Когда подходит |
| --- | --- | --- |
| `bar` — **Линия** (по умолчанию) | дорожка 40 × 3 на месте текста, отрезок пробегает с ускорением и торможением за 1.1 с | основной вариант: компактно, читается на любом фоне |
| `underline` — **Линия под текстом** | текст остаётся, под ним бежит полоска шириной с текст | когда в тексте важная информация (сумма, срок) |

| `spinner` — **Спиннер** | кольцо 3/4 из Figma, оборот за 0.8 с | текущий вариант в Figma |

Общее поведение:

- Ширина кнопки не меняется: текст остаётся в разметке. У `bar`, `dots` и `spinner` текст уходит в прозрачность с размытием 2 px и масштабом 0.96, лоадер проявляется с небольшой задержкой (200 мс, токен `--button-motion-loading-duration`). Обратный переход тот же.
- В загрузке кнопка не нажимается (`aria-disabled`), но остаётся в фокусе. Для скринридера `aria-busy="true"` и к имени добавляется «, Загрузка» (`loadingLabel`).
- Reduced motion: отрезок и блик не ездят, индикатор мягко «дышит» прозрачностью на месте. Спиннер вращается, как системный индикатор.

## Движение

- **Нажатие как у Button.** На `pointerdown` чипс сжимается до 0.96 и фон переходит в `control/bg/pressed` за 100 мс. Отпускание мягче, 240 мс. Выбор перекрашивает фон в те же 240 мс, поэтому нажатие и смена цвета читаются как одно движение.
- **Крестик на пружине.** При выборе крестик вырастает из 0.5 с небольшим перелётом (та же кривая, что у сердца Like) и выходит из размытия. Правый край чипса выезжает через `clip-path`, без анимации ширины.
- **Снятие выбора.** Сначала край заезжает обратно и крестик гаснет (180 мс), и только потом чипс становится уже. Поэтому ничего не прыгает.
- **Соседи доезжают.** В `ChipGroup` чипсы после изменившегося плавно переезжают на новое место, а при переносе на новую строку скользят туда же. Анимируется свойство `translate`, поэтому сжатие при нажатии продолжает работать.
- Всё прерывается: новый тап посреди анимации стартует с того места, где она сейчас.
- **Reduced motion:** без масштаба, без выезда края и переезда соседей. Крестик просто появляется и гаснет, цвета меняются как обычно.

## Анимация

Включите в тулбаре **Motion → Замедление ×4**, чтобы рассмотреть анимации по кадрам.

**1. Нажатие на кнопку** (CSS transition, `CounterButton.module.css`)

- Реакция начинается на `pointerdown`, а не на отпускание: белый круг сжимается 36 → 32 px (Figma State=Pressed), иконка остаётся 20 px.
- Нажатие быстрое (100 мс), отпускание мягче (240 мс), кривая strong ease-out. Цвета при переходе в disabled меняются за 160 мс.
- Это transition, а не keyframes: короткий тап разворачивает анимацию с того места, где круг сейчас. Если увести палец с кнопки, нажатие отменяется, как у нативной кнопки.
- Значение меняется на `click` (отпускание), как у нативного степпера.

**2. Смена цифры** (Web Animations API, `RollingNumber.tsx`)

- При увеличении старая цифра уходит вверх, новая приходит снизу; при уменьшении наоборот.
- Двигаются только изменившиеся разряды: 19 → 20 крутит оба, 20 → 21 только последний.
- **Барабан + blur — утверждённый вариант.** Цифры стоят на маленьком цилиндре радиусом 20 px и переворачиваются через верхний или нижний край поля, уходя в глубину. Пропа для выбора эффекта у Counter нет, это стандарт DS. В `RollingNumber` остался `effect="slide"` (прямой сдвиг) на будущее, например для цен.
- **Размытие в движении**: в центре цифра резкая, к краю размывается до 2 px. Скорость читается лучше, а две цифры сливаются в одно движение, а не выглядят двумя отдельными объектами.
- Новая цифра доворачивается за 300 мс, старая уходит за 260 мс. Кривая ease-out quart без перелёта: большая часть пути проходит за первые ~100 мс, остальное время уходит на мягкую посадку.
- Быстрые действия анимация не тормозит. Значение, `onChange` и озвучка обновляются сразу по тапу. Анимация только догоняет значение и прерывается в любой момент: новое значение стартует с того положения, где цифра сейчас, поэтому частые тапы и автоповтор раскручивают барабан, а не перезапускают его.
- **Разворот**: «+» и сразу «−» (1 → 2 → 1) не прокручивает вторую «1» с края. Уходящая «1» возвращается с того места, где она сейчас, а «2» уезжает туда, откуда пришла.
- Ширина разряда меряется в момент смены, а не берётся из кэша: веб-шрифт может догрузиться после первой отрисовки и поменять ширины цифр. Новый разряд (99 → 100) въезжает с нулевой ширины, а уходящий сжимается до нуля и не удаляется посреди анимации.
- Размываются только цифра 10 × 19 px и её «призрак», так что даже на слабом Android это дёшево. Если нужно, `--counter-motion-roll-blur: 0px` выключает размытие.
- Края поля мягкие за счёт маски, поэтому край плашки цифру не обрезает.

- В Inter Mbank нет моноширинных цифр (в шрифте нет OpenType-фич, `tnum` не работает): «1» шириной 7 px, «0» 10 px. Поэтому ширина разряда плавно подстраивается за то же время, а не прыгает. В одноразрядном счётчике цифра стоит в поле шириной 28 px, и кнопки не сдвигаются.
- Reduced motion: вместо движения короткая смена прозрачности на месте.

## Логика

- Значение всегда в пределах `min…max`, `onChange` получает уже обрезанное значение и не вызывается, если упёрлись в лимит.
- Удержание кнопки повторяет шаг, как `UIStepper` на iOS: первый повтор через 400 мс, дальше каждые 110 мс. На лимите повтор останавливается, `autoRepeat={false}` его выключает.
- Клавиатура на группе: ↑ / → прибавляют, ↓ / ← убавляют, Home и End прыгают к лимитам.
- Долгое нажатие не вызывает выделение текста и системное меню (`user-select`, `-webkit-touch-callout`, `contextmenu`). `touch-action: manipulation` убирает задержку двойного тапа.

## Доступность

- Корень `role="group"` с `aria-label` (например «Взрослые»), у кнопок свои имена через `decrementLabel` и `incrementLabel`.
- Значение лежит в `<output aria-live="polite">`, VoiceOver и TalkBack читают новое число после шага.
- На лимите кнопка `aria-disabled`, а не `disabled`: фокус не выпадает на `<body>`.
- Зона нажатия 44 × 44, видимый круг 36. Фокус с клавиатуры показывается кольцом `border/focus`.

## Анимации

- **Подпись** плывёт вверх: крупная подпись-плейсхолдер (IM-16 M) уезжает вверх и уменьшается, а маленькая (IM-12 M) проявляется на её месте за 220 мс. Это два слоя, каждый в своём начертании, и на стыке они чуть размываются (1.5 px), поэтому в покое текст всегда резкий.
- **Крестик «очистить»** появляется, когда в поле есть текст: вырастает с 0.5 до 1 с лёгким перелётом за 300 мс, как крестик в Chips. Уходит быстрее, за 180 мс. Место под ним зарезервировано, поэтому текст не прыгает. Это CSS-переход: если быстро печатать и стирать, анимация разворачивается с текущего места. Тап по крестику не уводит фокус, и клавиатура на телефоне не закрывается.
- **Подсказка и ошибка** раскрываются под полем по высоте за 240 мс, а не выскакивают за один кадр.
- **Select**: при нажатии фон за 100 мс переходит на `control/bg/pressed` и возвращается за 240 мс, как у кнопок. Когда список открыт, шеврон поворачивается вверх.
- **Reduced motion**: крестик только проявляется, без роста.

## Доступность

- Подпись Field — настоящий `<label for>`. Подсказка и ошибка связаны с полем через `aria-describedby`. При ошибке у поля `aria-invalid`, а текст ошибки объявляется как `role="alert"`.

- У Search роль `search`, а `placeholder` по умолчанию становится именем поля.
- Фокус в Field виден по курсору и подписи, которая уезжает вверх. Фон в фокусе не меняется, как в Figma, обводки нет — для клавиатуры это слабый сигнал, см. вопросы ниже.

## Использование

```tsx
import { IconButton, CloseIcon, ArrowLeftIcon } from '@mtravel/ui';
```

## Нажатие (предложение)

В Figma нет состояния Pressed, поэтому это предложение, оно ощущается так же, как у Button:

- на `pointerdown` кнопка сжимается до 0.94 за 100 мс (у Button 0.97, но маленькой кнопке нужно заметнее), на отпускании возвращается за 240 мс;
- Secondary заливается `control/bg/pressed`, Ghost получает `control/bg/default`, Primary остаётся стеклом (отвечает только масштаб);

- `disabled`: иконка `control/icon/disabled`, нажатия нет;
- `pressed` принудительно включает вид нажатия, только для доков и визуальных тестов;
- reduced motion: без масштаба, заливка всё равно меняется.

## Движение

- **Ползунок идёт за пальцем 1:1** с того места, где его взяли: он не прыгает центром под палец. Значение под ним прилипает к шагу.
- **Инерция.** Отпустили с разгону — ползунок едет дальше с той же скоростью и плавно тормозит: скорость каждую миллисекунду умножается на 0.99, путь примерно скорость × 100 мс. Потом он встаёт на ближайший шаг. Упрётся в край или в другой ползунок — остановится там. Отпустили медленно (палец стоял дольше 45 мс) — инерции нет, ползунок просто доезжает до шага за 240 мс. Новое касание ловит едущий ползунок там, где он сейчас.
- **Тап по треку** двигает ближний ползунок туда же, плавно, и с этого касания его можно сразу тянуть. Когда ползунки стоят друг на друге, ползунок выбирает направление первого движения: влево — «от», вправо — «до».
- **Нажатие** отвечает сразу, на касание: тень ползунка уходит за 100 мс, подсказка вырастает из своего кончика (0.8 → 1, 200 мс, ease-out).
- **Подсказка — шарик на нитке.** Кончик догоняет ползунок на пружине (response 200 мс, damping 0.86): на ходу чуть отстаёт, в конце мягко доезжает. Сам пузырь наклоняется назад, против движения, тем сильнее, чем быстрее тянут (0.02° на каждый px/с, не больше 14°). Когда ползунок встал, пузырь выпрямляется на мягкой пружине (damping 0.5) и чуть качается, как шарик. Ось наклона — кончик над ползунком.
- **Сумма меняется целиком** ([SwapNumber](?path=/docs/components-swapnumber--документация)): старое число уезжает по вертикали, размываясь и угасая, новое въезжает с другой стороны и становится чётким, за 140 мс. Посередине оба мягкие и сливаются, поэтому отдельные цифры не мелькают. Больше — новое снизу, меньше — сверху. Следующая смена начинается, когда предыдущая доехала, а последнее значение показывается всегда: пока ползунок едет, сумма меняется несколько раз в секунду, а не каждый кадр.
- **Отпустили** — сумма видна ещё 400 мс после того, как ползунок встал, и уходит быстрее, чем появилась (150 мс). `onChangeEnd` приходит, когда ползунок остановился, а не в момент отпускания.
- **Клавиатура:** стрелки двигают ползунок сразу, без доезда, а подсказка догоняет его на пружине. Если клавишу держат, ползунок не отстаёт.
- **Reduced motion:** нет пружины, наклона, инерции и доезда. Подсказка и сумма меняются прозрачностью, ползунок встаёт на место сразу.

## Движение

Повторяет модель Sonner из статьи Эмиля Ковальского [«Building a toast component»](https://emilkowal.ski/ui/building-a-toast-component), только сверху вниз:

- **Въезд сверху.** Новый снекбар выезжает из-за верхнего края (`translateY(-100%)` → 0) за 400 мс по CSS `ease` и проявляется. Медленнее кнопок: уведомление не отвечает на касание, ему можно быть спокойнее.
- **Стопка.** Новый всегда спереди. Те, что позади, выглядывают снизу на 14 px и становятся на 5 % меньше каждый. Все они той же высоты, что передний, и без текста, поэтому край стопки ровный. Видно три, остальные ждут за ними.
- **Раскрытие.** Наведи курсор или тапни по стопке: снекбары расходятся друг под другом, каждый сдвигается на сумму высот выше и зазоры по 8 px. Промежутки между ними заполнены, поэтому наведение не теряется. Тап мимо стопки сворачивает её.
- **Переходы, а не keyframes.** Если быстро нажать несколько раз, снекбары доезжают на новые места с того положения, где они сейчас, без рывков.
- **Свайп вверх закрывает.** Дальше 45 px или быстрым броском (скорость больше 0.11 px/мс) на любой дистанции. Вниз тянется с трением и возвращается. Кнопка внутри нажимается как обычно.
- **Отпускание быстрее въезда.** Въезд спокойный: 400 мс по `ease`. Всё, что отвечает на палец, идёт за 240 мс по сильному ease-out: возврат на место, если не дотянул, и уход после броска. Бросок не проваливается в медленный старт, снекбар улетает с той же скоростью, с какой его кинули.
- **Один палец.** Свайп ведёт только тот палец, который его начал. Второй палец на снекбаре не перехватывает жест, и снекбар не прыгает.
- **Таймер.** 4 секунды, на паузе, пока стопка раскрыта, палец на снекбаре или вкладка скрыта. Оставшееся время сохраняется.
- **Reduced motion:** ничего не едет, снекбары проявляются и гаснут на месте.

## Движение

- **Целиком, а не по разрядам.** Старое число уезжает по вертикали на 8 px, размываясь до 3 px и угасая. Новое въезжает с другой стороны, из размытия в чёткость. Посередине оба мягкие и сливаются в одно движение, отдельные цифры не мелькают.

- **Ширина** плавно подстраивается под новое число: в Inter Mbank цифры разной ширины.

/** Reads --sheet-motion-* from CSS on the sheet itself, so Storybook's slow-motion / reduced-motion reach the JS. */
function readConfig(el: HTMLElement): MotionConfig {
  const m = readMotion(el);
  const num = (token: string, fallback: number) => {
    const n = parseFloat(m.value(token, ''));
    return Number.isFinite(n) ? n : fallback;
  };
  return {
    openDuration: m.duration('--sheet-motion-open-duration', DEFAULT_CONFIG.openDuration),
    closeDuration: m.duration('--sheet-motion-close-duration', DEFAULT_CONFIG.closeDuration),
    easing: parseCubicBezier(m.value('--sheet-motion-easing', '')) ?? DEFAULT_CONFIG.easing,
    springResponse: m.duration('--sheet-motion-spring-response', DEFAULT_CONFIG.springResponse),
    springDamping: num('--sheet-motion-spring-damping', DEFAULT_CONFIG.springDamping),
    flickDamping: num('--sheet-motion-flick-damping', DEFAULT_CONFIG.flickDamping),
    exitAccel: num('--sheet-motion-exit-accel', DEFAULT_CONFIG.exitAccel),
    exitSpeed: num('--sheet-motion-exit-speed', DEFAULT_CONFIG.exitSpeed),
    scale: num('--motion-scale', 1),
    fadeDuration: m.duration('--sheet-motion-fade-duration', DEFAULT_CONFIG.fadeDuration),

 *
 * Motion lives in `engine.ts` (tween on the drawer curve from rest, a velocity-carrying spring for anything
 * the user touched) and `gesture.ts` (1:1 drag, scroll hand-off, rubber band, flick). React only decides
 * when the sheet is mounted: it stays in the tree while it slides out and unmounts when it has left.
 */
export function BottomSheet({
  open,
  defaultOpen = false,
  onOpenChange,
  header,
  title,
  description,
  'aria-label': ariaLabel,
  showHandle = true,
  showClose = true,
  closeLabel = 'Закрыть',
  dismissible = true,
  primaryAction,
  secondaryAction,
  footer,
  initialFocus,
  children,
  className,
  onOpened,
  onClosed,
}: BottomSheetProps) {

  const [present, setPresent] = useState(isOpen);
  if (isOpen && !present) setPresent(true);

  const titleId = useId();
  const sheetRef = useRef<HTMLDivElement>(null);
  const backdropRef = useRef<HTMLDivElement>(null);
  const closeRef = useRef<HTMLButtonElement>(null);
  const contentRef = useRef<HTMLDivElement>(null);
  const motionRef = useRef<SheetMotion | null>(null);
  const [overflowing, setOverflowing] = useState(false);

  // always-current values for the long-lived engine and gesture callbacks
  const live = useRef({ isOpen, dismissible, onOpened, onClosed, setOpen });
  live.current = { isOpen, dismissible, onOpened, onClosed, setOpen };

  const resolvedHeader: BottomSheetHeader = header ?? (title ? 'left' : 'none');
  const hasFooter = footer != null || primaryAction != null || secondaryAction != null;

  // Closing starts inside the same event (no wait for React): the sheet moves on the very next frame.
  // A controlled parent may refuse; two frames later a sheet that is still "open" reverses from where it is.
  const requestClose = useCallback(() => {
    if (!live.current.isOpen) return;
    const m = motionRef.current;
    const sheet = sheetRef.current;
    if (m && sheet && m.goal === 'open') {
      m.cfg = readConfig(sheet);
      m.go('closed');
    }
    live.current.setOpen(false);
    requestAnimationFrame(() =>
      requestAnimationFrame(() => {
        const cur = motionRef.current;
        if (cur && sheet && live.current.isOpen && cur.goal === 'closed') {
          cur.cfg = readConfig(sheet);
          cur.go('open');
        }
      }),
    );
  }, []);

  /* ---------- motion: engine + gesture, created when the sheet is mounted ---------- */
  useLayoutEffect(() => {
    if (!present) return;
    const sheet = sheetRef.current;
    const backdrop = backdropRef.current;
    if (!sheet || !backdrop) return;

    const m = new SheetMotion(sheet, backdrop, readConfig(sheet));
    motionRef.current = m;
    m.H = sheet.offsetHeight;
    m.reset(true);
    m.onSettle = (goal) => {
      if (goal === 'open') {
        live.current.onOpened?.();
      } else if (live.current.isOpen) {
        // closed by a gesture, but the parent kept it open: come back
        m.cfg = readConfig(sheet);
        m.go('open');
      } else {
        motionRef.current = null;
        setPresent(false);
        live.current.onClosed?.();
      }

      motion: () => motionRef.current,
      dismissible: () => live.current.dismissible,
      isOpen: () => live.current.isOpen,
      requestClose,
      requestOpen: () => live.current.setOpen(true),
      refresh: () => {
        if (motionRef.current) motionRef.current.cfg = readConfig(sheet);
      },
    });

    // content height changes, rotation, keyboard: the cached height is the only thing the frame loop reads
    const ro = new ResizeObserver(() => m.setHeight(sheet.offsetHeight));
    ro.observe(sheet);

    // a scrollable slot must be reachable from the keyboard: it becomes a focusable region only while it overflows
    const content = contentRef.current;
    const measure = () => content && setOverflowing(content.scrollHeight > content.clientHeight + 1);
    const ro2 = new ResizeObserver(measure);
    const mo = new MutationObserver(() => {
      measure();
      if (content) for (const c of Array.from(content.children)) ro2.observe(c);
    });
    if (content) {
      ro2.observe(content);
      for (const c of Array.from(content.children)) ro2.observe(c);
      mo.observe(content, { childList: true });
    }
    measure();

    m.go(live.current.isOpen ? 'open' : 'closed');

    return () => {
      ro2.disconnect();
      mo.disconnect();
      ro.disconnect();
      gesture.destroy();
      m.destroy();
      if (motionRef.current === m) motionRef.current = null;
    };

  // open <-> closed from the outside (props, ✕, Escape, backdrop): reverses from wherever the sheet is
  useLayoutEffect(() => {
    const m = motionRef.current;
    const sheet = sheetRef.current;

    if ((m.goal === 'open') === isOpen) return;

      inertRef.current = null;
      uninert();
      unlock();
    };

  useEffect(() => {
    if (!isOpen || !present) return;
    const sheet = sheetRef.current;
    const root = sheet?.parentElement;
    if (!sheet || !root) return;

    const trigger = document.activeElement instanceof HTMLElement ? document.activeElement : null;
    const id = Symbol('sheet');
    sheetStack.push(id);

    const target =
      typeof initialFocus === 'string'
        ? initialFocus === 'close'
          ? closeRef.current
          : sheet.querySelector<HTMLElement>(initialFocus)
        : initialFocus?.current;
    (target ?? sheet).focus({ preventScroll: true });

    const onKey = (e: globalThis.KeyboardEvent) => {
      if (e.key === 'Escape' && sheetStack.isTop(id) && live.current.dismissible && !e.defaultPrevented) {
        e.preventDefault();
        requestClose();
      }
    };
    const onFocusIn = (e: FocusEvent) => {
      if (sheetStack.isTop(id) && e.target instanceof Node && !root.contains(e.target)) (sheet.querySelector<HTMLElement>(FOCUSABLE) ?? sheet).focus({ preventScroll: true });
    };
    document.addEventListener('keydown', onKey);
    document.addEventListener('focusin', onFocusIn);

    return () => {
      document.removeEventListener('keydown', onKey);
      document.removeEventListener('focusin', onFocusIn);
      sheetStack.remove(id);
      if (trigger && trigger.isConnected && trigger !== document.body) {
        // the page stays inert until the sheet has left; the one thing that must be focusable now is the trigger
        const top = trigger.closest('body > *') as (HTMLElement & { inert?: boolean }) | null;
        if (top && top !== root) {
          top.inert = false;
          top.removeAttribute('aria-hidden');
        }
        trigger.focus({ preventScroll: true });
      }
    };
  }, [isOpen, present, initialFocus, requestClose]);

    if (e.key !== 'Tab') return;
    const sheet = sheetRef.current;
    if (!sheet) return;
    const items = Array.from(sheet.querySelectorAll<HTMLElement>(FOCUSABLE)).filter((el) => el.tabIndex >= 0 && !el.hasAttribute('inert'));
    const first = items[0];
    const last = items[items.length - 1];
    if (!first || !last) {
      e.preventDefault();
      return;
    }
    const active = document.activeElement;
    if (e.shiftKey && (active === first || active === sheet)) {
      e.preventDefault();
      last.focus();
    } else if (!e.shiftKey && active === last) {
      e.preventDefault();
      first.focus();
    }
  };

  if (!present || typeof document === 'undefined') return null;

  const closeSize = resolvedHeader === 'none' ? 40 : 36;
  const named = resolvedHeader !== 'none' && title ? { 'aria-labelledby': titleId } : { 'aria-label': ariaLabel ?? (typeof title === 'string' ? title : undefined) };

  return createPortal(
    <div className={styles.root} data-state={isOpen ? 'open' : 'closed'}>
      <div
        ref={backdropRef}
        className={styles.backdrop}
        style={{ opacity: 0 }}
        onClick={() => live.current.dismissible && requestClose()}
        aria-hidden="true"
        data-sheet-backdrop=""
      />
      <div
        ref={sheetRef}
        className={[styles.sheet, className].filter(Boolean).join(' ')}
        style={{ transform: 'translate3d(0, 100%, 0)' }}
        role="dialog"
        aria-modal="true"
        tabIndex={-1}
        onKeyDown={onKeyDown}
        data-header={resolvedHeader}
        data-handle={showHandle || undefined}
        data-close={showClose || undefined}
        data-footer={hasFooter || undefined}
        {...named}
      >
        <div className={styles.panel}>
          {showHandle && (
            <div className={styles.handle} aria-hidden="true">
              <span className={styles.bar} />
            </div>
          )}
          {showClose && (
            <IconButton
              ref={closeRef}
              className={styles.close}
              icon={<CloseIcon />}
              aria-label={closeLabel}
              variant="ghost"
              radius={16}
              size={closeSize}
              onClick={requestClose}
            />
          )}
          {resolvedHeader === 'none' ? (
            showClose && <div className={styles.topbar} aria-hidden="true" />
          ) : (
            <header className={styles.header}>
              <div className={styles.headerRow}>
                {title && (
                  <h2 id={titleId} className={`im-24-b ${styles.title}`}>
                    {title}
                  </h2>
                )}
                {description && <p className={`im-16-r ${styles.description}`}>{description}</p>}
              </div>
            </header>

            ref={contentRef}
            className={styles.content}
            data-sheet-content=""
            {...(overflowing
              ? { tabIndex: 0, role: 'region', 'aria-label': typeof title === 'string' ? title : (ariaLabel ?? 'Содержимое') }
              : {})}
          >
            {children}

                      {secondaryAction.label}
                    </Button>
                  )}
                </>
              )}
            </footer>
          )}
        </div>
      </div>
    </div>,
    document.body,
  );
}

## Движение

Всё движение шторки считает собственный движок (`engine.ts`, `gesture.ts`), без зависимостей. CSS-анимации для жестов не годятся: их нельзя схватить и развернуть
на лету. Шторка двигается только через `transform` и `opacity`, `transform` пишется прямо в элемент (не через CSS-переменную, чтобы не пересчитывать стили всего содержимого),
`will-change` стоит только пока что-то движется.

### Модель

Положение `y` в px от открытого состояния: `0` открыта, `H` целиком ниже экрана (`H` — высота шторки, кэшируется и обновляется через `ResizeObserver`).
Прозрачность фона `= 1 − clamp(y / H, 0, 1)` пересчитывается при каждой записи положения, поэтому фон всегда привязан к шторке.

| Ситуация | Что двигает шторку | Почему |
| --- | --- | --- |
| Открытие из покоя | кривая drawer `cubic-bezier(0.32, 0.72, 0, 1)`, 500 мс | резкий старт и длинная мягкая посадка, как у iOS; вход — ease-out |
| Закрытие (✕, фон, Escape, `open=false`) | та же кривая назад, 400 мс | тот же путь, но быстрее: система отвечает, пользователь не ждёт |
| Уход после жеста, разворот открытия в закрытие | кубическая кривая Эрмита из (y, v₀) в (H, 1500 px/с) с ограниченным стартовым ускорением | шторка продолжает на скорости пальца и набирает скорость плавно (рост не больше ~4 px за кадр на кадр). Пружина, нацеленная на край, давала 100 000 px/с² с первого кадра и бросала шторку быстрее пальца |
| Возврат в открытое положение, захват движущейся шторки | пружина с начальной скоростью пальца | у пружины нет «рестарта», она подхватывает положение и скорость |

Анимация по кривой стартует с экранного положения. Если шторка уже летит быстрее 150 px/с (разворот, захват), берётся Эрмит (на уход) или пружина (на возврат), чтобы скорость не обрывалась. Закрытие начинается синхронно в обработчике ✕, фона и Escape, без ожидания React.

### Уход после жеста

```
y(u) = h00·y₀ + h10·T·v₀ + h01·H + h11·T·v_end,   u = t / T
y''(0) = (6D − T(4v₀ + 2v_end)) / T² = 14 400 px/с²,   D = H − y₀,  v_end = max(v₀, 1500 px/с)
```

`T` решает это уравнение и не превышает 400 мс. Токены `--sheet-motion-exit-accel`, `--sheet-motion-exit-speed`. Если шторка летела вверх (разворот), она сначала гасит эту скорость с тем же ускорением и затем идёт вниз.

### Пружина возврата

Параметры как у Apple: отклик (response) и коэффициент демпфирования (damping ratio), решение в замкнутом виде, поэтому результат не зависит от частоты кадров.

```
ω  = 2π / response           response = 380 мс (токен --sheet-motion-spring-response)
ζ  = 1                       без отскока (по умолчанию)
ζ  = 0.82                    после броска вверх: лёгкий отскок
ζ = 1:  x(t) = (A + (v₀ + ωA)·t) · e^(−ωt)
ζ < 1:  x(t) = e^(−ζωt) · (A·cos ω_d t + (v₀ + ζωA)/ω_d · sin ω_d t),  ω_d = ω·√(1 − ζ²)
```

`A` — расстояние до цели в момент старта, `v₀` — скорость пальца в px/с.

### Что решает при отпускании

- Скорость берётся по последним ~100 мс движения шторки. Если с последнего `touchmove` прошло больше **45 мс** (палец постоял), скорость равна нулю: iOS не шлёт события для неподвижного пальца.
- **Бросок** (скорость вниз ≥ **0.3 px/мс**) закрывает после 8 px пути. **Мягкий бросок** (0.11–0.3 px/мс) закрывает после 48 px. Медленнее — решает только расстояние: дальше **30 %** высоты, либо проекция инерции `y + v/1000 · 0.99/(1 − 0.99)` уходит за 30 %. Движение вверх быстрее 0.11 px/мс всегда возвращает шторку.
- Возврат идёт критически демпфированной пружиной. После броска вверх добавляется отскок `ζ = 0.82`: перелёт есть, потому что у жеста был импульс.

### Резинка

Вверх от открытого положения шторка идёт с растущим сопротивлением: `f(x) = x·L·k / (L + k·x)`, `L = 120 px` (потолок), `k = 0.55` (начальное сопротивление).
Порог начала жеста вычитается, поэтому 100 px пальца дают около трети (33–37 px) движения шторки, жёсткой остановки нет. Щель под приподнятой шторкой закрывает продолжение фона шторки. Если `dismissible=false`, то же сопротивление работает и вниз.

### Reduced motion

При `prefers-reduced-motion` (или «Motion → Reduced motion» в тулбаре) открытие и закрытие становятся короткой сменой прозрачности шторки и фона без
перемещения: длительность **200 мс** по ease-out, заметная часть (от 2 до 98 %) занимает около 170 мс. Перетаскивание остаётся 1:1, потому что его ведёт сам пользователь; короткий жест возвращает шторку пружиной, длинный закрывает её плавным исчезновением.
«Замедление ×4» из тулбара умножает все длительности, в том числе отклик пружины.

## Жесты

| Жест | Поведение |
| --- | --- |
| Тянуть за ручку, заголовок, футер или любую неподвижную область | шторка идёт за пальцем 1:1, с тем смещением, где схватили. Начало жеста — после **4 px** на неподвижных поверхностях и **10 px** внутри прокручиваемого контента, поэтому тапы по кнопкам работают |
| Тянуть по прокручиваемому контенту вниз | пока контент не в самом верху, он прокручивается; дойдя до верха, тот же жест продолжает тянуть шторку. Дальше назад вверх шторка сначала возвращается, потом снова прокручивается контент |
| Тянуть по контенту вверх | прокручивается контент, шторка стоит |
| Горизонтальное движение | игнорируется (карусели внутри контента) |
| Второй палец | игнорируется: ни начать жест, ни перехватить его нельзя |
| Тронуть движущуюся шторку (не кнопку, ссылку или поле: тап по ✕ шторку не останавливает) | она останавливается под пальцем с текущего экранного положения и идёт за ним; если палец поднят без движения, шторка продолжает путь к прежней цели с прежней скоростью |
| Закрывается, а пользователь тянет назад | шторка разворачивается из текущей точки, `onOpenChange(true)` сообщает родителю |
| Мышь и стилус | Pointer Events с `setPointerCapture`; после перетаскивания клик под курсором не срабатывает |

Для пальца используются touch events, а не pointer events: браузер отменяет pointer-события в момент, когда сам начинает прокрутку, и передать жест от прокрутки
шторке внутри одного касания было бы нельзя. Касание неявно захвачено элементом, на котором началось (это эквивалент `setPointerCapture`), а `preventDefault()`
на `touchmove` не даёт странице и WKWebView прокручиваться за шторкой.

## Доступность

- `role="dialog"`, `aria-modal="true"`, имя из заголовка (`aria-labelledby`) или из `aria-label`.
- Слот с прокруткой получает `tabIndex=0`, `role="region"` и `aria-label` только пока контент не помещается, чтобы его можно было прокручивать стрелками; порядок Tab не меняется, фокус при открытии он не забирает.
- Фокус уходит в шторку при открытии (на саму шторку, либо на `initialFocus`) и возвращается на элемент, открывший её, при закрытии.
- Tab и Shift+Tab ходят по кругу внутри шторки; всё остальное на странице получает `inert` и `aria-hidden`. ✕ первый в порядке обхода.
- Escape закрывает (если `dismissible`). Верхняя из нескольких шторок получает Escape первой.
- Блокировка скролла и `inert` снимаются, когда шторка уехала (перед размонтированием, в том числе при размонтировании во время закрытия), а фокус возвращается сразу в начале закрытия.
- Страница под шторкой не прокручивается: блокировка скролла `position: fixed` на `body`, она же держит iOS Safari, позиция прокрутки восстанавливается.
- Кнопка ✕ доступна всегда, даже при `dismissible={false}`.

## Известные ограничения

- Проверено в Chromium с эмуляцией касаний. Ощущение на реальном iPhone зависит от WebKit: проверьте на устройстве, как ведёт себя резинка прокрутки внутри слота.
  `overscroll-behavior` в iOS старше 16 не поддерживается, поэтому там внутренний контент может «пружинить» при прокрутке до верха одновременно с шторкой.
- Chrome подавляет первые ~15 px `touchmove` (touch slop): на Android шторка начинает идти за пальцем с первого доставленного события, смещение остаётся постоянным.
- Шторка не поднимается выше своей высоты: детентов (medium / large) нет.

# Motion

В Figma нет переменных для длительности и кривых, поэтому motion-токены живут в коде: `src/tokens/motion.css`. Они устроены так же, как цвета: базовая шкала, а поверх неё токены компонента.

## Правила

1. **Анимировать, только если есть зачем.** Анимация подтверждает нажатие, показывает смену состояния или объясняет, откуда что взялось. Если действие повторяется сотни раз в день, анимации нет.
2. **Реакция на касание, а не на отпускание.** Обратная связь стартует на `pointerdown` и укладывается в 100–160 мс.
3. **UI-анимации короче 300 мс.** Уход быстрее появления, отпускание мягче нажатия.
4. **Ease-out для всего, что запускает пользователь.** `ease-in` не используем: он медленно стартует ровно тогда, когда человек смотрит.
5. **Двигаем только `transform` и `opacity`.** Они не трогают layout и идут на GPU, что важно в WebView на слабых Android.
6. **Анимацию можно прервать.** Новое действие посреди анимации стартует с текущего положения на экране: CSS transitions или WAAPI от presentation-значения, без keyframes с нуля.
7. **Reduced motion — это не отсутствие обратной связи.** Движение меняем на короткую смену прозрачности, цвет и прозрачность оставляем.

## Как пользоваться в коде

```css
.thing {
  transition: transform calc(var(--motion-duration-fast) * var(--motion-scale)) var(--motion-ease-out);
}
```

Длительности в CSS всегда умножаются на `--motion-scale`, в JS читаются через `readMotion(el)` из `src/lib/motion.ts`. Тогда переключатель **Motion → Замедление ×4** в тулбаре замедляет всё сразу, и анимацию можно разобрать покадрово.
